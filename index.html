-- MINECRAFT OBBY -- Place this Script inside ServerScriptService local Players = game:GetService("Players") local Workspace = game:GetService("Workspace") local Lighting = game:GetService("Lighting") local old = Workspace:FindFirstChild("MinecraftObby") if old then old:Destroy() end local folder = Instance.new("Folder") folder.Name = "MinecraftObby" folder.Parent = Workspace -- WORLD SETTINGS Workspace.Gravity = 196.2 Lighting.ClockTime = 14 Lighting.Brightness = 2 Lighting.Ambient = Color3.fromRGB(160, 160, 160) local checkpoints = {} local playerStage = {} local debounce = {} local finishDebounce = {} -- BLOCK CREATOR local function makeBlock(name, size, position, color, material) local part = Instance.new("Part") part.Name = name part.Size = size part.Position = position part.Anchored = true part.TopSurface = Enum.SurfaceType.Smooth part.BottomSurface = Enum.SurfaceType.Smooth part.Color = color part.Material = material or Enum.Material.SmoothPlastic part.Parent = folder return part end local GRASS = Color3.fromRGB(95, 159, 53) local DIRT = Color3.fromRGB(115, 78, 48) local STONE = Color3.fromRGB(110, 110, 110) local WOOD = Color3.fromRGB(126, 85, 50) local GOLD = Color3.fromRGB(255, 195, 40) -- Leaderboard local function setupPlayer(player) local stats = Instance.new("Folder") stats.Name = "leaderstats" stats.Parent = player local stage = Instance.new("IntValue") stage.Name = "Stage" stage.Value = 0 stage.Parent = stats local wins = Instance.new("IntValue") wins.Name = "Wins" wins.Value = 0 wins.Parent = stats playerStage[player] = 0 player.CharacterAdded:Connect(function(character) local root = character:WaitForChild("HumanoidRootPart", 10) if not root then return end task.wait(0.15) local savedStage = playerStage[player] or 0 local target = checkpoints[savedStage] if target and target.Parent then character:PivotTo(target.CFrame + Vector3.new(0, 4, 0)) end end) end Players.PlayerAdded:Connect(setupPlayer) for _, player in ipairs(Players:GetPlayers()) do task.spawn(setupPlayer, player) end Players.PlayerRemoving:Connect(function(player) playerStage[player] = nil debounce[player] = nil finishDebounce[player] = nil end) -- STARTING ISLAND makeBlock( "StartingGrass", Vector3.new(18, 2, 18), Vector3.new(0, 0, 0), GRASS ) makeBlock( "StartingDirt", Vector3.new(18, 3, 18), Vector3.new(0, -2.5, 0), DIRT ) local spawn = Instance.new("SpawnLocation") spawn.Name = "ObbySpawn" spawn.Size = Vector3.new(6, 1, 6) spawn.Position = Vector3.new(0, 1.5, 0) spawn.Anchored = true spawn.Neutral = true spawn.Color = Color3.fromRGB(80, 220, 80) spawn.Parent = folder checkpoints[0] = spawn -- LAVA UNDER THE COURSE local lava = makeBlock( "DangerLava", Vector3.new(240, 2, 28), Vector3.new(125, -7, 0), Color3.fromRGB(255, 65, 15), Enum.Material.Neon ) lava.Touched:Connect(function(hit) local character = hit:FindFirstAncestorOfClass("Model") local humanoid = character and character:FindFirstChildOfClass("Humanoid") if humanoid then humanoid.Health = 0 end end) -- CHECKPOINT TOUCH local function makeCheckpoint(part, stageNumber) checkpoints[stageNumber] = part part.Touched:Connect(function(hit) local character = hit:FindFirstAncestorOfClass("Model") local player = character and Players:GetPlayerFromCharacter(character) if not player then return end if debounce[player] then return end debounce[player] = true local current = playerStage[player] or 0 if stageNumber > current then playerStage[player] = stageNumber local stats = player:FindFirstChild("leaderstats") local stage = stats and stats:FindFirstChild("Stage") if stage then stage.Value = stageNumber end end task.delay(0.7, function() debounce[player] = nil end) end) end -- COURSE: 15 JUMP PLATFORMS for i = 1, 15 do local x = i * 12 local y = 0 if i % 3 == 0 then y = 3 elseif i % 3 == 1 then y = 0 else y = 1.5 end local color = (i % 2 == 0) and GRASS or STONE makeBlock( "JumpBlock" .. i, Vector3.new(8, 2, 8), Vector3.new(x, y, 0), color ) makeBlock( "DirtUnder" .. i, Vector3.new(8, 1, 8), Vector3.new(x, y - 1.5, 0), DIRT ) -- Checkpoint every five platforms if i % 5 == 0 then local cp = makeBlock( "Checkpoint" .. i, Vector3.new(5, 0.5, 5), Vector3.new(x, y + 1.3, 0), Color3.fromRGB(30, 255, 90), Enum.Material.Neon ) cp.CanCollide = false makeCheckpoint(cp, i) end end -- SIMPLE BLOCKY TREES ON THE SIDES for i = 1, 5 do local x = i * 30 makeBlock( "TreeTrunk" .. i, Vector3.new(3, 7, 3), Vector3.new(x, 3.5, 12), WOOD ) makeBlock( "TreeLeaves" .. i, Vector3.new(9, 8, 9), Vector3.new(x, 9, 12), GRASS ) end -- FINISH PLATFORM local finishBase = makeBlock( "FinishIsland", Vector3.new(18, 2, 18), Vector3.new(204, 0, 0), GRASS ) makeBlock( "FinishDirt", Vector3.new(18, 3, 18), Vector3.new(204, -2.5, 0), DIRT ) local finish = makeBlock( "FINISH", Vector3.new(8, 1, 8), Vector3.new(204, 1.5, 0), GOLD, Enum.Material.Neon ) local sign = makeBlock( "FinishSign", Vector3.new(10, 6, 1), Vector3.new(204, 6, -8), GOLD ) local surfaceGui = Instance.new("SurfaceGui") surfaceGui.Face = Enum.NormalId.Front surfaceGui.Parent = sign local label = Instance.new("TextLabel") label.Size = UDim2.fromScale(1, 1) label.BackgroundTransparency = 1 label.Text = "YOU WIN!" label.TextScaled = true label.Font = Enum.Font.GothamBlack label.TextColor3 = Color3.fromRGB(40, 40, 40) label.Parent = surfaceGui finish.Touched:Connect(function(hit) local character = hit:FindFirstAncestorOfClass("Model") local player = character and Players:GetPlayerFromCharacter(character) if not player or finishDebounce[player] then return end finishDebounce[player] = true local stats = player:FindFirstChild("leaderstats") local wins = stats and stats:FindFirstChild("Wins") if wins then wins.Value += 1 end playerStage[player] = 0 local stage = stats and stats:FindFirstChild("Stage") if stage then stage.Value = 0 end task.delay(2, function() finishDebounce[player] = nil end) player:LoadCharacter() end) -- FALL RESET local fallPlane = makeBlock( "FallReset", Vector3.new(300, 1, 100), Vector3.new(100, -20, 0), Color3.fromRGB(255, 0, 0) ) fallPlane.Transparency = 1 fallPlane.CanCollide = false fallPlane.Touched:Connect(function(hit) local character = hit:FindFirstAncestorOfClass("Model") local humanoid = character and character:FindFirstChildOfClass("Humanoid") if humanoid then humanoid.Health = 0 end end) print("Minecraft Obby loaded successfully!")
